# Multilanguage (en/it/fr/de/es) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The desktop app UI, user-visible backend errors, and chat answer language work in English, Italian, French, German and Spanish, chosen in Settings (default: OS language).

**Architecture:** Frontend owns all translation (`i18next` + JSON catalogs). ragcore (Python) and the Rust shell emit stable error `code`s that the UI maps to translated strings, falling back to the raw English message. Chat requests carry `lang`; ragcore appends an "Answer in <Language>" line to the system prompt.

**Tech Stack:** React 19, Vite, Vitest, i18next, react-i18next, Tauri 2 (Rust), FastAPI/pydantic.

**Spec:** `docs/superpowers/specs/2026-09-29-i18n-design.md`

## Global Constraints

- Supported language codes, exactly: `en`, `it`, `fr`, `de`, `es`. Stored preference values: `system | en | it | fr | de | es` in `localStorage` key `language`.
- `en` is the source-of-truth catalog; fallback language is `en`.
- Log lines (Python `logger.*`, Rust `supervisor.log`) stay English. Do not translate them.
- Unmigrated backend errors must keep working (string `detail` still shown raw).
- Catalog parity: every locale has exactly `en`'s key set and identical `{{placeholder}}` names per key.
- Translation style: infinitive/imperative labels, avoid tu/vous/du/Sie/tú address.
- After every completed task run `scripts/install-local.sh` (project memory rule). Do not cut a release.
- Do not touch unrelated files already modified in the working tree (`git status` shows pre-existing edits); `git add` only the files each task names.
- Frontend commands run from `apps/desktop` (`bun`); Python from `core/ragcore`.

## Review Focus

- OS language `pt-BR` / `zh` / empty `navigator.language` → `en`, never a crash or a missing-key string like `chat.title`.
- Region tags (`it-IT`, `de-AT`, `fr_CA`) resolve to their base language.
- Stored `language` value that is garbage or a removed language → treated as `system`.
- Backend error with a `code` the UI has no key for → raw English `message` shown, not the key.
- Old ragcore string `detail` and Rust `"404: model not installed"` style errors still display verbatim.
- `lang` missing or `en` on `/query` → system prompt byte-identical to today's.
- German/French strings longer than English overflow tight layouts (icon rail, buttons, sidebar) — checked in manual pass.
- Slide converter has its own `language: "it"|"en"` field (`SlideConversionRequest`); it must not silently change meaning (see Task 6).

- [ ] **Step 5: Implement `src/lib/i18n.ts`**

```ts
import i18n from "i18next";
import { initReactI18next } from "react-i18next";

import de from "@/locales/de.json";
import en from "@/locales/en.json";
import es from "@/locales/es.json";
import fr from "@/locales/fr.json";
import it from "@/locales/it.json";

export const LANGUAGES = ["en", "it", "fr", "de", "es"] as const;
export type Language = (typeof LANGUAGES)[number];
export type LanguagePref = "system" | Language;

const STORAGE_KEY = "language";

const isLanguage = (value: unknown): value is Language =>
  typeof value === "string" && (LANGUAGES as readonly string[]).includes(value);

export function resolveLanguage(pref: unknown, navigatorLang: string | undefined): Language {
  if (isLanguage(pref)) return pref;
  const base = (navigatorLang ?? "").split(/[-_]/)[0].toLowerCase();
  return isLanguage(base) ? base : "en";
}

export function getLanguagePref(): LanguagePref {
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    return isLanguage(stored) ? stored : "system";
  } catch {
    return "system";
  }
}

export function currentLanguage(): Language {
  return resolveLanguage(getLanguagePref(), globalThis.navigator?.language);
}

export function setLanguagePref(pref: LanguagePref): void {
  try {
    if (pref === "system") localStorage.removeItem(STORAGE_KEY);
    else localStorage.setItem(STORAGE_KEY, pref);
  } catch {
    // Private mode: the choice lasts until reload.
  }
  const language = currentLanguage();
  document.documentElement.lang = language;
  void i18n.changeLanguage(language);
}

void i18n.use(initReactI18next).init({
  resources: {
    en: { translation: en },
    it: { translation: it },
    fr: { translation: fr },
    de: { translation: de },
    es: { translation: es },
  },
  lng: currentLanguage(),
  fallbackLng: "en",
  interpolation: { escapeValue: false }, // React already escapes.
  returnNull: false,
});
if (typeof document !== "undefined") document.documentElement.lang = currentLanguage();

export default i18n;
```

import { describe, expect, it } from "vitest";
import de from "./de.json";
import en from "./en.json";
import es from "./es.json";
import fr from "./fr.json";
import it_ from "./it.json";

type Tree = { [key: string]: string | Tree };

function flatten(tree: Tree, prefix = ""): Record<string, string> {
  return Object.entries(tree).reduce<Record<string, string>>((out, [key, value]) => {
    const path = prefix ? `${prefix}.${key}` : key;
    return typeof value === "string"
      ? { ...out, [path]: value }
      : { ...out, ...flatten(value, path) };
  }, {});
}

const placeholders = (text: string) => [...text.matchAll(/{{\s*(\w+)\s*}}/g)].map((m) => m[1]).sort();
const base = flatten(en as Tree);

describe.each([["it", it_], ["fr", fr], ["de", de], ["es", es]] as const)("%s catalog", (_name, catalog) => {
  const flat = flatten(catalog as Tree);
  it("has exactly the keys of en", () => {
    expect(Object.keys(flat).sort()).toEqual(Object.keys(base).sort());
  });
  it("keeps the same placeholders", () => {
    for (const [key, text] of Object.entries(base)) {
      expect(placeholders(flat[key] ?? ""), key).toEqual(placeholders(text));
    }
  });
  it("has no empty strings", () => {
    for (const [key, text] of Object.entries(flat)) expect(text.trim(), key).not.toBe("");
  });
});

import type { TFunction } from "i18next";
import i18n from "@/lib/i18n";
import { IpcError } from "@/lib/ipc";

export function parseIpcError(raw: string): IpcError {
  try {
    const parsed: unknown = JSON.parse(raw);
    if (parsed && typeof parsed === "object" && typeof (parsed as { code?: unknown }).code === "string") {
      const { code, message, params } = parsed as { code: string; message?: string; params?: Record<string, unknown> };
      return new IpcError(message ?? code, code, params ?? {});
    }
  } catch {
    // Not JSON: an old-style string error.
  }
  return new IpcError(raw);
}

export function errorText(
  err: unknown,
  t: TFunction | ((key: string, opts?: Record<string, unknown>) => string) = i18n.t.bind(i18n),
  exists: (key: string) => boolean = (key) => i18n.exists(key),
): string {
  if (err instanceof IpcError && err.code && exists(`errors.${err.code}`)) {
    return t(`errors.${err.code}`, err.params ?? {}) as string;
  }
  return err instanceof Error ? err.message : String(err);
}
